Full Item Template ​

This is the full copy-paste reference for item config. Copy only the fields you need into a business file.

For what each field does, see the Complete Field Reference.

lua
-- Full og_consumable item template
-- Copy only the fields you need. This file is reference material, not a file loaded by fxmanifest.

local business = 'example_business'

OGConsumable.RegisterCategory(business, 'example_category', 'food', {
    example_full_item = {
        label = 'Example Full Item',
        hunger = 50,
        thirst = 10,
        stress = 5,
        log = 'You used an Example Full Item.',
        progress = 'Using Example Item...',
        duration = 6,
        remove = true,
        count = 1,
        dirtyItem = 'example_dirty_plate',
        dirtyCount = 1,
        dirtyMetadata = true,
        requiredItems = {
            { item = 'example_spoon', count = 1 }
        },
        restrictions = {
            mode = 'any',
            whitelist = false,
            ace = 'og_consumable.admin',
            citizenids = {
                'ABC12345'
            },
            peds = {
                'a_c_chop'
            },
            message = 'You cannot use this item.'
        },
        cooldown = 2500,
        propPreset = 'food',
        anim = {
            dict = 'mp_player_inteat@burger',
            clip = 'mp_player_int_eat_burger',
            flag = 49
        },
        effects = {
            health = 10,
            armor = 5,
            stamina = {
                restore = 25,
                duration = 8000,
                interval = 1000
            }
        },
        alcohol = {
            provider = 'nrg',
            drunk = 0.20,
            notify = false
        },
        ox = {
            description = 'A complete example of what an og_consumable item can do.',
            weight = 200,
            image = 'example_full_item.png',
            stack = true,
            close = true,
            consume = 0,
            degrade = 1440,
            decay = true,
            export = 'og_consumable.UseConsumable'
        }
    }
})
